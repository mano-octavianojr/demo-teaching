\begin{frame}{The learning goal}
\begin{block}{Our example}
Four fictional student records become a saved class summary.
\end{block}
\begin{itemize}
\item Represent each student using a name and grade.
\item Calculate the average and identify passing students.
\item Save the results in a text file.
\end{itemize}
\end{frame}
\begin{frame}[fragile]{A loop processes the grades}
\begin{lstlisting}
grades = [88, 72, 95, 75]
total = 0
for grade in grades:
    total += grade
print(total / len(grades))
\end{lstlisting}
\textbf{Expected output:} 82.5
\end{frame}
\begin{frame}{Quick check}
When does the loop run its indented body?
\begin{enumerate}
\item Once for the entire list.
\item Once for each item in the list.
\end{enumerate}
\medskip
\textbf{Answer:} once per item, for four iterations here.
\end{frame}
